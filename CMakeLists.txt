cmake_minimum_required(VERSION 3.25)
set(AUDIO_ENGINE_REQUESTED_BUILD_TYPE "${CMAKE_BUILD_TYPE}")
project(AudioEngine VERSION 0.1.0 LANGUAGES C CXX)

if(PROJECT_IS_TOP_LEVEL
   AND NOT CMAKE_CONFIGURATION_TYPES
   AND "${AUDIO_ENGINE_REQUESTED_BUILD_TYPE}" STREQUAL "")
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

if(PROJECT_IS_TOP_LEVEL)
  include(CTest)
  set(AUDIO_ENGINE_TESTS_DEFAULT ${BUILD_TESTING})
else()
  set(AUDIO_ENGINE_TESTS_DEFAULT OFF)
endif()

option(AUDIO_ENGINE_BUILD_TESTS "Build the audio_engine tests" ${AUDIO_ENGINE_TESTS_DEFAULT})
option(AUDIO_ENGINE_BUILD_EXAMPLES "Build the audio_engine examples" ${PROJECT_IS_TOP_LEVEL})
option(AUDIO_ENGINE_ENABLE_ASIO "Build the Windows x86-64 ASIO backend" OFF)
set(AUDIO_ENGINE_ASIO_SDK "" CACHE PATH "External ASIO SDK root containing common/iasiodrv.h")

function(audio_engine_set_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 $<$<COMPILE_LANGUAGE:CXX>:/permissive->)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
  endif()
endfunction()

add_library(audio_engine STATIC src/engine.cpp src/sample_conversion.cpp)
add_library(audio_engine::audio_engine ALIAS audio_engine)
target_compile_features(audio_engine PRIVATE cxx_std_20)
target_include_directories(audio_engine PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
set_target_properties(audio_engine PROPERTIES CXX_EXTENSIONS OFF)
audio_engine_set_warnings(audio_engine)

if(AUDIO_ENGINE_ENABLE_ASIO)
  if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "AMD64|amd64|x86_64|X86_64")
    message(FATAL_ERROR "ASIO currently requires Windows x86-64")
  endif()
  if(NOT EXISTS "${AUDIO_ENGINE_ASIO_SDK}/common/iasiodrv.h")
    message(FATAL_ERROR "Set AUDIO_ENGINE_ASIO_SDK to your external ASIO SDK directory")
  endif()
  target_sources(audio_engine PRIVATE src/backends/windows/asio_backend.cpp)
  target_include_directories(audio_engine PRIVATE src)
  target_include_directories(audio_engine SYSTEM PRIVATE "${AUDIO_ENGINE_ASIO_SDK}/common")
  target_compile_definitions(audio_engine PRIVATE AUDIO_ENGINE_HAS_ASIO=1 NOMINMAX WIN32_LEAN_AND_MEAN)
  target_link_libraries(audio_engine PRIVATE ole32 advapi32 user32)
endif()

if(AUDIO_ENGINE_BUILD_TESTS)
  enable_testing()
  add_subdirectory(unittests)
endif()

if(AUDIO_ENGINE_BUILD_EXAMPLES)
  add_subdirectory(examples)
endif()

include(tools/DeveloperTools.cmake)
